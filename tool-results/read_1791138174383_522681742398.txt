    87→    87→function trim1(v: number): string {
    88→    88→  return v % 1 === 0 ? v.toFixed(0) : v.toFixed(1);
    89→    89→}
    90→    90→
    91→    91→function fmt(n: number): string {
    92→    92→  if (n >= 100_000) return `${trim1(n / 1000)}K`;
    93→    93→  return n.toLocaleString("en-US");
    94→    94→}
    95→    95→
    96→    96→function kfmt(v: number): string {
    97→    97→  return v >= 1000 ? `${trim1(v / 1000)}K` : `${v}`;
    98→    98→}

   140→   140→function rateTone(rate: number): { bar: string; text: string; label: string } {
   141→   141→  const pct = rate * 100;
   142→   142→  if (pct < 2) return { bar: "bg-emerald-500", text: "text-emerald-400", label: "مستقر" };
   143→   143→  if (pct <= 3) return { bar: "bg-amber-500", text: "text-amber-400", label: "متقلب نوعًا" };
   144→   144→  return { bar: "bg-rose-500", text: "text-rose-400", label: "متقلب" };
   145→   145→}

   158→   158→function avatarGrad(handle: string): string {
   159→   159→  let h = 0;
   160→   160→  for (let i = 0; i < handle.length; i++) h = (h * 31 + handle.charCodeAt(i)) >>> 0;

   731→   731→export default function DashboardSection() {
   732→   732→  type FetchState =
   733→   733→    | { status: "loading" }
   734→   734→    | { status: "error"; message: string }
   735→   735→    | { status: "ok"; data: ContributionsData };
   736→   736→
   737→   737→  const [state, setState] = useState<FetchState>({ status: "loading" });
   738→   738→  const [reloadKey, setReloadKey] = useState(0);
   739→   739→
   740→   740→  const retry = () => {
   741→   741→    setState({ status: "loading" });
   742→   742→    setReloadKey((k) => k + 1);
   743→   743→  };
   744→   744→
   745→   745→  useEffect(() => {
   746→   746→    const ctrl = new AbortController();
   747→   747→
   748→   748→    fetch("/api/contributions", { signal: ctrl.signal })
   749→   749→      .then(async (res) => {
   750→   750→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   751→   751→        return (await res.json()) as ContributionsData;
   752→   752→      })
   753→   753→      .then((d) => setState({ status: "ok", data: d }))
   754→   754→      .catch((e: unknown) => {
   755→   755→        if (e instanceof DOMException && e.name === "AbortError") return;
   756→   756→        setState({ status: "error", message: e instanceof Error ? e.message : "network error" });
   757→   757→      });
   758→   758→
   759→   759→    return () => ctrl.abort();
   760→   760→  }, [reloadKey]);
   761→   761→
   762→   762→  return (
   763→   763→    <section aria-label="لوحة بيانات المساهمات" className="mx-auto w-full max-w-7xl px-4 py-16 sm:py-20">
   764→   764→      <SectionHeader
   765→   765→        badge="بيانات تجريبية"
   766→   766→        title="لوحة بيانات المساهمات"
   767→   767→        description="نموذج واقعي لنشاط مستودع pytorch/pytorch: كوميتات أسبوعية، توزيع المشاكل، تغطية الاختبارات، لوحة صدارة المساهمين، واختبارات متقلبة — أرقام تجريبية مولّدة بأنماط قريبة من الواقع، وليست بيانات حية من GitHub."
   768→   768→        icon={BarChart3}
   769→   769→      />
   770→   770→
   771→   771→      {state.status === "loading" && <DashboardSkeleton />}
   772→   772→      {state.status === "error" && <ErrorState message={state.message} onRetry={retry} />}
   773→   773→      {state.status === "ok" && <DashboardContent data={state.data} />}
   774→   774→    </section>
   775→   775→  );
   776→   776→}
