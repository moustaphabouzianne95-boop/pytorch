   442→   442→export default function ApiExplorerSection() {
   443→   443→  const [status, setStatus] = useState<FetchStatus>("loading");
   444→   444→  const [groups, setGroups] = useState<DocGroup[] | null>(null);
   445→   445→  const [errorMsg, setErrorMsg] = useState<string | null>(null);
   446→   446→  const [reloadToken, setReloadToken] = useState(0);
   447→   447→
   448→   448→  const [query, setQuery] = useState("");
   449→   449→  const [activeGroup, setActiveGroup] = useState<string>("all");
   450→   450→  /** undefined = derive (first result) · null = explicitly closed (mobile) · string = explicit pick */
   451→   451→  const [manualId, setManualId] = useState<string | null | undefined>(undefined);
   452→   452→
   453→   453→  const inputRef = useRef<HTMLInputElement>(null);
   454→   454→  const listRef = useRef<HTMLDivElement>(null);
   455→   455→
   456→   456→  /* Fetch docs from the REST API (state updates only inside async callbacks) */
   457→   457→  useEffect(() => {
   458→   458→    let cancelled = false;
   459→   459→    fetch("/api/docs")
   460→   460→      .then(async (res) => {
   461→   461→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   462→   462→        const data = (await res.json()) as { groups?: DocGroup[] };
   463→   463→        if (!data || !Array.isArray(data.groups) || data.groups.length === 0) {
   464→   464→          throw new Error("bad payload");
   465→   465→        }
   466→   466→        if (!cancelled) {
   467→   467→          setGroups(data.groups);
   468→   468→          setStatus("ready");
   469→   469→        }
   470→   470→      })
   471→   471→      .catch(() => {
   472→   472→        if (!cancelled) {
   473→   473→          setErrorMsg(
   474→   474→            "تعذّر تحميل مرجع API من الخادم. تحقق من الاتصال ثم أعد المحاولة."

   476→   476→          setStatus("error");
   477→   477→        }
   478→   478→      });
   479→   479→    return () => {
   480→   480→      cancelled = true;
   481→   481→    };
   482→   482→  }, [reloadToken]);
   483→   483→
   484→   484→  /* "/" focuses the search box (unless already typing in a field) */

   490→   490→        (t.tagName === "INPUT" ||

   499→   499→    return () => window.removeEventListener("keydown", onKey);

   501→   501→
   502→   502→  /* Flatten groups once data arrives */

   519→   519→        f.entry.signature.toLowerCase().includes(q) ||

   522→   522→    });
   523→   523→  }, [flat, activeGroup, query]);
   524→   524→
   525→   525→  /* Displayed entry: explicit pick wins; otherwise fall back to the first result.

   530→   530→  );
   531→   531→
   532→   532→  /* Scroll the master list back to top when the scope changes */
   533→   533→  useEffect(() => {
   534→   534→    listRef.current?.scrollTo({ top: 0 });
   535→   535→  }, [activeGroup, query]);
   536→   536→
   537→   537→  const groupTabs = useMemo(
   538→   538→    () => [

   541→   541→        id: g.id,
   542→   542→        name: g.name,
   543→   543→        count: g.entries.length,
   544→   544→      })),
   545→   545→    ],
   546→   546→    [groups, flat.length]

   548→   548→
   549→   549→  const activeGroupDesc =
   550→   550→    activeGroup === "all"

   568→   568→    setManualId(undefined);
   569→   569→  };

   571→   571→  return (

   573→   573→      {/* Soft ambient glow */}

   576→   576→        className="pointer-events-none absolute -top-32 start-1/3 h-80 w-80 rounded-full bg-primary/10 blur-[110px]"
   577→   577→      />
   578→   578→
   579→   579→      <div className="relative mx-auto w-full max-w-7xl px-4">

   581→   581→        <motion.div {...fadeUp}>
   582→   582→          <SectionHeader
   583→   583→            badge="مرجع قابل للبحث"

   586→   586→            icon={BookOpen}
   587→   587→          />
   588→   588→        </motion.div>
   589→   589→
   590→   590→        {/* Search bar */}

   594→   594→              className="pointer-events-none absolute start-4 top-1/2 h-5 w-5 -translate-y-1/2 text-muted-foreground"
   595→   595→              aria-hidden
   596→   596→            />
   597→   597→            <Input
   598→   598→              ref={inputRef}

   638→   638→        </motion.div>
   639→   639→
   640→   640→        {/* Group selector with counts */}
   641→   641→        <motion.div {...fadeUp} className="mb-6 flex flex-wrap items-center gap-2">
   642→   642→          {groupTabs.map((t) => {

   644→   644→            const latin = isLatin(t.name);

   751→   751→                      <MousePointerClick className="h-7 w-7" aria-hidden />

   757→   757→                  </div>

   759→   759→              </aside>

   761→   761→
   762→   762→            {/* Mobile: accordion cards */}
   763→   763→            <div className="flex flex-col gap-3 lg:hidden">
   764→   764→              {filtered.map((f) => (
   765→   765→                <MobileEntryCard
   766→   766→                  key={f.entry.name}
   767→   767→                  f={f}
   768→   768→                  open={f.entry.name === manualId}
   769→   769→                  onToggle={() =>
