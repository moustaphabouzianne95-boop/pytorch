   442→export default function ApiExplorerSection() {
   443→  const [status, setStatus] = useState<FetchStatus>("loading");
   444→  const [groups, setGroups] = useState<DocGroup[] | null>(null);
   445→  const [errorMsg, setErrorMsg] = useState<string | null>(null);
   446→  const [reloadToken, setReloadToken] = useState(0);
   447→
   448→  const [query, setQuery] = useState("");
   449→  const [activeGroup, setActiveGroup] = useState<string>("all");
   450→  /** undefined = derive (first result) · null = explicitly closed (mobile) · string = explicit pick */
   451→  const [manualId, setManualId] = useState<string | null | undefined>(undefined);
   452→
   453→  const inputRef = useRef<HTMLInputElement>(null);
   454→  const listRef = useRef<HTMLDivElement>(null);
   455→
   456→  /* Fetch docs from the REST API (state updates only inside async callbacks) */
   457→  useEffect(() => {
   458→    let cancelled = false;
   459→    fetch("/api/docs")
   460→      .then(async (res) => {
   461→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   462→        const data = (await res.json()) as { groups?: DocGroup[] };
   463→        if (!data || !Array.isArray(data.groups) || data.groups.length === 0) {
   464→          throw new Error("bad payload");
   465→        }
   466→        if (!cancelled) {
   467→          setGroups(data.groups);
   468→          setStatus("ready");
   469→        }
   470→      })
   471→      .catch(() => {
   472→        if (!cancelled) {
   473→          setErrorMsg(
   474→            "تعذّر تحميل مرجع API من الخادم. تحقق من الاتصال ثم أعد المحاولة."
   475→          );
   476→          setStatus("error");
   477→        }
   478→      });
   479→    return () => {
   480→      cancelled = true;
   481→    };
   482→  }, [reloadToken]);
   483→
   484→  /* "/" focuses the search box (unless already typing in a field) */
   485→  useEffect(() => {
   486→    const onKey = (e: KeyboardEvent) => {
   487→      const t = e.target as HTMLElement | null;
   488→      const typing =
   489→        t &&
   490→        (t.tagName === "INPUT" ||
   491→          t.tagName === "TEXTAREA" ||
   492→          t.isContentEditable);
   493→      if (e.key === "/" && !typing) {
   494→        e.preventDefault();
   495→        inputRef.current?.focus();
   496→      }
   497→    };
   498→    window.addEventListener("keydown", onKey);
   499→    return () => window.removeEventListener("keydown", onKey);
   500→  }, []);
   501→
   502→  /* Flatten groups once data arrives */
   503→  const flat = useMemo<FlatEntry[]>(
   504→    () =>
   505→      (groups ?? []).flatMap((g) =>
   506→        g.entries.map((e) => ({ entry: e, groupId: g.id, groupName: g.name }))
   507→      ),
   508→    [groups]
   509→  );
   510→
   511→  /* Instant search: name + signature + Arabic description */
   512→  const filtered = useMemo(() => {
   513→    const q = query.trim().toLowerCase();
   514→    return flat.filter((f) => {
   515→      if (activeGroup !== "all" && f.groupId !== activeGroup) return false;
   516→      if (!q) return true;
   517→      return (
   518→        f.entry.name.toLowerCase().includes(q) ||
   519→        f.entry.signature.toLowerCase().includes(q) ||
   520→        f.entry.description.includes(query.trim())
   521→      );
   522→    });
   523→  }, [flat, activeGroup, query]);
   524→
   525→  /* Displayed entry: explicit pick wins; otherwise fall back to the first result.
   526→     Pure derivation — no selection effect needed. */
   527→  const active = useMemo(
   528→    () => filtered.find((f) => f.entry.name === manualId) ?? filtered[0] ?? null,
   529→    [filtered, manualId]
   530→  );
   531→
   532→  /* Scroll the master list back to top when the scope changes */
   533→  useEffect(() => {
   534→    listRef.current?.scrollTo({ top: 0 });
   535→  }, [activeGroup, query]);
   536→
   537→  const groupTabs = useMemo(
   538→    () => [
   539→      { id: "all", name: "الكل", count: flat.length },
   540→      ...(groups ?? []).map((g) => ({
   541→        id: g.id,
   542→        name: g.name,
   543→        count: g.entries.length,
   544→      })),
   545→    ],
   546→    [groups, flat.length]
   547→  );
   548→
   549→  const activeGroupDesc =
   550→    activeGroup === "all"
   551→      ? null
   552→      : (groups ?? []).find((g) => g.id === activeGroup)?.description;
   553→
   554→  const clearSearch = () => {
   555→    setQuery("");
   556→    setManualId(undefined);
   557→    inputRef.current?.focus();
   558→  };
   559→
   560→  const retry = () => {
   561→    setManualId(undefined);
   562→    setStatus("loading");
   563→    setReloadToken((t) => t + 1);
   564→  };
   565→
   566→  const selectGroup = (id: string) => {
   567→    setActiveGroup(id);
   568→    setManualId(undefined);
   569→  };
   570→
   571→  return (
   572→    <section className="relative overflow-hidden py-20" aria-label="مرجع PyTorch API">
   573→      {/* Soft ambient glow */}
